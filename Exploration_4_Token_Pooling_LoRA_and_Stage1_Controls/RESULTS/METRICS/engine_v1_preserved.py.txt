def load_data():
 rows=read_manifest(OLD/'01_sample_definition/sample_manifest.csv');cv=np.array([int(r['cv_fold']) for r in csv.DictReader((OLD/'01_sample_definition/prompt3_cv.csv').open())]);y=torch.tensor([r['label'] for r in rows]);return rows,cv,y
def norm(x):return nn.functional.layer_norm(torch.as_tensor(np.asarray(x).copy(),dtype=torch.float32),(1536,))
class Model(nn.Module):
 def __init__(self,extra=0):
  super().__init__();self.base=Architecture('A5',16);self.extra=nn.ModuleList()
  for _ in range(extra):
   v=nn.Sequential(nn.Linear(1536,8,bias=False),nn.Linear(8,10,bias=False));nn.init.zeros_(v[-1].weight);self.extra.append(v)
 def forward(self,x):
  z=self.base(x[:,:1552]);self.corrections=[]
  for k,layer in enumerate(self.extra):
   d=layer(x[:,1552+k*1536:1552+(k+1)*1536]);z=z+d;self.corrections.append(d.detach().norm(dim=1).mean().item())
  return z
def representations():
 dest=P/'TARGET_POOLING/REPRESENTATIONS';dest.mkdir(exist_ok=True)
 if (dest/'complete.json').exists():return
 rows,cv,y=load_data();patch=np.load(P/'TOKEN_AUDIT/CACHE/patch.npy',mmap_mode='r');cls=np.load(OLD/'02_cache/uni2_features/gt_fov96/features.npy');n=len(rows)
 coords=np.array([[(float(r[k])-np.floor(float(r[k])-48+.5))/6 for k in ['x','y']] for r in rows])
 broken=np.random.default_rng(1701).uniform(2,14,(n,2));shuffled=coords[np.random.default_rng(1701).permutation(n)]
 grid=np.stack(np.meshgrid(np.arange(16)+.5,np.arange(16)+.5),-1).reshape(-1,2)
 outputs={k:np.empty((n,1536),dtype='float32') for k in ['A1','A2','A3','A1_broken','A2_broken','A3_broken','A2_literal_shuffle','A2_offcenter']};weights={}
 for label,centers in [('real',coords),('broken',broken),('literal_shuffle',shuffled),('offcenter',coords+4)]:
  ws=[]
  for i in range(n):
   xy=centers[i];w=np.exp(-((grid-xy)**2).sum(1)/(2*1.5**2));w=w/w.sum();ws.append(w)
   g=(patch[i]*w[:,None]).sum(0)
   if label in ['real','broken']:
    suffix='' if label=='real' else '_broken';cx,cy=np.clip(np.floor(xy).astype(int),0,15);outputs['A1'+suffix][i]=patch[i,cy*16+cx]
    near=(np.abs(np.arange(256)%16-cx)<=1)&(np.abs(np.arange(256)//16-cy)<=1);outputs['A3'+suffix][i]=patch[i,near].mean(0);outputs['A2'+suffix][i]=g
   else:outputs['A2_'+label][i]=g
  weights[label]=np.array(ws,dtype='float32')
 for suffix in ['', '_broken']:outputs['A4'+suffix]=((norm(cls)+norm(outputs['A2'+suffix]))/2).numpy()
 for k,v in outputs.items():np.save(dest/(k+'.npy'),v)
 np.savez_compressed(dest/'pooling_geometry.npz',coordinates=coords,broken=broken,literal_shuffle=shuffled,grid=grid,**weights)
 dump(dest/'complete.json',{'rows':n,'files':{k:sha(dest/(k+'.npy')) for k in outputs},'coordinate_patch_range':[coords.min(0).tolist(),coords.max(0).tolist()],'literal_shuffle_max_shift_patches':float(np.linalg.norm(coords-shuffled,axis=1).max()),'broken_not_GT_dependent':True})
def inputs(cfg,ti):
 rep=cfg.get('representation','A0');h=norm(np.load(OLD/'02_cache/uni2_features/gt_fov96/features.npy') if rep=='A0' else np.load(P/'TARGET_POOLING/REPRESENTATIONS'/(rep+'.npy')))
 b=np.load(OLD/'02_cache/biology/tierA.npy');mean=b[ti].mean(0);std=np.maximum(b[ti].std(0),1e-6);b=torch.from_numpy(((b-mean)/std).astype('float32'));parts=[h,b]
 for scale in cfg.get('scales',[]):parts.append(norm(np.load(OLD/f'02_cache/uni2_features/gt_fov{scale}/features.npy')))
 return torch.cat(parts,1),{'mean':mean.tolist(),'std':std.tolist()}
def extra_metrics(rows,y,z):
 m=summary_metrics(rows,y,z);cm=np.array(m['confusion']);per=[]
 for c,name in enumerate(CLASSES):
  fn=cm[c].copy();fn[c]=0;fp=cm[:,c].copy();fp[c]=0
  per.append({'class':name,'positive_ROIs':len({r['roi'] for r,t in zip(rows,y) if t==c}),'semantic_TP':int(cm[c,c]),'semantic_FP':int(cm[:,c].sum()-cm[c,c]),'semantic_FN':int(cm[c].sum()-cm[c,c]),'dominant_FN_class':CLASSES[fn.argmax()] if fn.sum() else None,'dominant_FP_source':CLASSES[fp.argmax()] if fp.sum() else None})
 m['class_diagnostics']=per;return m
